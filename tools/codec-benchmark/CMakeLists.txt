cmake_minimum_required(VERSION 3.25)
project(LAN3DCodecBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(CodecBenchmark ../CodecBenchmark.cpp ../../src/media/Codec.cpp ../../src/media/NvEncoder.cpp ../../src/media/NvDecoder.cpp)
target_include_directories(CodecBenchmark PRIVATE ../../src ../../third_party/nvcodec)
target_compile_definitions(CodecBenchmark PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(CodecBenchmark PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(CodecBenchmark PRIVATE mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 advapi32 d3d11 dxgi)
